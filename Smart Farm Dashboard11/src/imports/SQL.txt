-- 1. settings 테이블 생성
CREATE TABLE IF NOT EXISTS settings (
    id INT PRIMARY KEY DEFAULT 1,
    target_stage INT DEFAULT 2,          -- 청경채 성장 단계 (1~6)
    led_brightness INT DEFAULT 80,       -- LED 밝기 (%)
    target_temp NUMERIC DEFAULT 21.5,    -- 목표 온도 (°C)
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. 기본 데이터 1건 삽입
INSERT INTO settings (id, target_stage, led_brightness, target_temp)
VALUES (1, 2, 80, 21.5)
ON CONFLICT (id) DO NOTHING;

-- 3. ★ 중요: Supabase Realtime 활성화 (테이블 변경을 앱으로 실시간 전송)
ALTER PUBLICATION supabase_realtime ADD TABLE settings;

-- ───────────── sensor_logs (센서 기록 그래프) ─────────────
create table if not exists public.sensor_logs (
  id bigint generated always as identity primary key,
  created_at timestamptz not null default now(),
  water_level real, soil_moisture real, light_lux real, co2_ppm real,
  target_temp real, humidity real, led_brightness real
);
create index if not exists sensor_logs_created_at_idx on public.sensor_logs (created_at);
alter table public.sensor_logs enable row level security;
create policy "sensor_logs read" on public.sensor_logs for select using (true);
create policy "sensor_logs insert" on public.sensor_logs for insert with check (true);
